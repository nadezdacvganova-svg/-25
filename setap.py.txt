@contextmanager
def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    try:
        yield conn

@app.get("/api/projects")
async def get_projects():
    with get_db() as conn:
        # Получаем проекты
        projects = conn.execute("SELECT * FROM projects").fetchall()
        
        # Если проектов нет, создаем демо
        if not projects:
            conn.execute("INSERT INTO projects (id, name) VALUES (?, ?)", ("demo_1", "Проект Альфа"))
            conn.execute("INSERT INTO projects (id, name) VALUES (?, ?)", ("demo_2", "Проект Бета"))
            conn.commit()
            projects = conn.execute("SELECT * FROM projects").fetchall()
            
            # Добавляем демо-транзакции
            demo_trans = [
                ("demo_1", "income", "Продажа", 150000, "Оплата по договору"),
                ("demo_1", "expense", "Внешние программисты", 45000, "Разработка модуля"),
                ("demo_2", "expense", "Аренда сервера", 12000, "Аренда VPS"),
            ]
            for trans in demo_trans:
                conn.execute("""
                    INSERT INTO transactions (project_id, type, category, amount, description)
                    VALUES (?, ?, ?, ?, ?)
                """, trans)
            conn.commit()
        
        result = []
        for project in projects:
            # Получаем транзакции проекта
            transactions = conn.execute(
                "SELECT * FROM transactions WHERE project_id = ?", (project["id"],)
            ).fetchall()
            
            total_income = sum(t["amount"] for t in transactions if t["type"] == "income")
            total_expenses = sum(t["amount"] for t in transactions if t["type"] == "expense")
            profit = total_income - total_expenses
            profitability = (profit / total_income * 100) if total_income > 0 else 0
            
            income_by_category = {}
            expenses_by_category = {}
            for t in transactions:
                if t["type"] == "income":
                    income_by_category[t["category"]] = income_by_category.get(t["category"], 0) + t["amount"]
                else:
                    expenses_by_category[t["category"]] = expenses_by_category.get(t["category"], 0) + t["amount"]
            
            # Получаем участников
            members = conn.execute(
                "SELECT * FROM project_members WHERE project_id = ?", (project["id"],)
            ).fetchall()
            
            result.append({
                "project_id": project["id"],
                "project_name": project["name"],
                "total_income": total_income,
                "total_expenses": total_expenses,
                "profit": profit,
                "profitability": profitability,
                "income_by_category": income_by_category,
                "expenses_by_category": expenses_by_category,
                "transactions": [dict(t) for t in transactions],
                "members": [{"user_id": m["user_id"], "user_name": m["user_name"]} for m in members]
            })
        
        return result

@app.post("/api/transactions")
async def add_transaction(transaction: TransactionCreate):
    with get_db() as conn:
        # Проверяем существование проекта
        project = conn.execute("SELECT * FROM projects WHERE id = ?", (transaction.project_id,)).fetchone()
        if not project:
            raise HTTPException(status_code=404, detail="Проект не найден")
        
        cursor = conn.execute("""
            INSERT INTO transactions (project_id, type, category, amount, description, employee_id)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (transaction.project_id, transaction.type, transaction.category, 
              transaction.amount, transaction.description, transaction.employee_id))
        conn.commit()
        
        return {"id": cursor.lastrowid, "message": "Транзакция добавлена"}

@app.get("/api/categories")
async def get_categories():
    with get_db() as conn:
        categories = conn.execute("SELECT type, name FROM categories ORDER BY type, name").fetchall()
        result = {"income": [], "expense": []}
        for cat in categories:
            result[cat["type"]].append(cat["name"])
        return result

@app.post("/api/projects")
async def create_project(project: ProjectCreate):
    with get_db() as conn:
        # Проверяем, не существует ли проект
        existing = conn.execute("SELECT * FROM projects WHERE id = ?", (project.id,)).fetchone()
        if existing:
            raise HTTPException(status_code=400, detail="Проект с таким ID уже существует")
        
        conn.execute("INSERT INTO projects (id, name) VALUES (?, ?)", (project.id, project.name))
        conn.commit()
        return {"message": "Проект создан"}

        async function syncProjects() {
            try {
                const response = await fetch('/api/projects');
                if (!response.ok) throw new Error('Ошибка загрузки');
                projects = await response.json();
                renderProjects(projects);
                updateProjectSelects();
            } catch (error) {
                document.getElementById('projectsList').innerHTML =

            }
        }

        function renderProjects(projects) {

            if (!projects || projects.length === 0) {

                return;

                    <h3>${p.project_name}</h3>
                    <div class="finance-grid">
                        <div class="finance-item"><div class="label">Доходы</div><div class="value positive">${formatCurrency(p.total_income)} ₽</div></div>
                        <div class="finance-item"><div class="label">Расходы</div><div class="value negative">${formatCurrency(p.total_expenses)} ₽</div></div>
                        <div class="finance-item"><div class="label">Прибыль</div><div class="value profit">${formatCurrency(p.profit)} ₽</div></div>

        async function addTransaction() {

                project_id: document.getElementById('transProjectId').value,
                type: document.getElementById('transType').value,

                amount: parseFloat(document.getElementById('transAmount').value),

                employee_id: null
            };
            
            if (!data.project_id || !data.category || !data.amount) {
                alert('Заполните все обязательные поля');
                return;
            }

            try {
                const response = await fetch('/api/transactions', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(data)
                });
                if (!response.ok) throw new Error('Ошибка');
                alert('✅ Транзакция добавлена!');
                closeModal('transactionModal');
                syncProjects();
                document.getElementById('transactionForm').reset();
            } catch (error) {
                alert('❌ Ошибка: ' + error.message);
            }
        }
